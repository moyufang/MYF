
# Axios

安装
```
# 使用 npm
npm install axios

# 使用 yarn
yarn add axios

# 使用 CDN
<script src="https://cdn.jsdelivr.net/npm/axios/dist/axios.min.js"></script>

// 在项目中引入
import axios from 'axios';
```

### 基本使用

GET 请求
```javascript
// 带参数的 GET 请求
axios.get('/api/users', {
  params: {
    page: 1,
    limit: 10,
    search: 'john'
  }
}).then(response => {
    console.log(response.data);
  })
  .catch(error => {
    console.error('Error:', error);
  });

// 使用 async/await
async function getUsers() {
  try {
    const response = await axios.get('/api/users');
    return response.data;
  } catch (error) {
    console.error('Error fetching users:', error);
  }
}

```
POST 请求
```javascript
// 创建新用户
axios.post('/api/users', {
  name: 'John Doe',
  email: 'john@example.com',
  age: 30
})
.then(response => {
  console.log('User created:', response.data);
});
```
PUT/PATCH 请求
```javascript
// PUT - 替换整个资源
axios.put('/api/users/1', {
  name: 'John Smith',
  email: 'johnsmith@example.com',
  age: 31
});

// PATCH - 部分更新资源
axios.patch('/api/users/1', {
  age: 32
});
```
DELETE 请求
```javascript
// 删除用户
axios.delete('/api/users/1')
  .then(response => {
    console.log('User deleted');
  });

// 带请求体的 DELETE（不常见，但支持）
axios.delete('/api/users/1', {
  data: {
    reason: 'inactive user'
  }
});
```
完整配置发起请求
```javascript
const config = {
  url: '/api/users', // 请求URL
  method: 'get', // 默认 GET
  baseURL: 'https://api.example.com', // 基础URL
  headers: {// 请求头
    'Content-Type': 'application/json',
    'Authorization': 'Bearer ' + token
  },
  params: {// 请求参数（GET请求）
    page: 1,
    limit: 10
  },
  data: {// 请求数据（POST/PUT/PATCH）
    name: 'John',
    email: 'john@example.com'
  },
  timeout: 5000,// 超时时间（毫秒）
  responseType: 'json', /// 响应类型，默认 json
  withCredentials: false, // 跨域请求是否携带凭据，默认 false
  adapter: null,// 请求适配器（用于Node.js或自定义）
  auth: {// 身份验证
    username: 'janedoe',
    password: 's00pers3cret'
  },
  responseEncoding: 'utf8', // 响应编码，moren
  xsrfCookieName: 'XSRF-TOKEN', // XSRF token 相关
  xsrfHeaderName: 'X-XSRF-TOKEN',
  // 上传/下载进度
  onUploadProgress: function(progressEvent) {
    const percentCompleted = Math.round(
      (progressEvent.loaded * 100) / progressEvent.total
    );
    console.log(`Upload: ${percentCompleted}%`);
  },
  onDownloadProgress: function(progressEvent) {
    const percentCompleted = Math.round(
      (progressEvent.loaded * 100) / progressEvent.total
    );
    console.log(`Download: ${percentCompleted}%`);
  }
};

// 使用配置发起请求
axios(config);
```


创建可持久化实例
```javascript
// 创建带有默认配置的实例
const apiClient = axios.create({
  baseURL: 'https://api.example.com',
  timeout: 10000,
  headers: {
    'Content-Type': 'application/json'
  }
});

// 使用实例
apiClient.get('/users')
  .then(response => console.log(response.data));

// 为特定实例添加配置
apiClient.defaults.headers.common['Authorization'] = 'Bearer ' + token;
```

### 拦截器
请求/响应拦截器
```javascript
// 添加请求拦截器
axios.interceptors.request.use(
  function (config) {
    // 在发送请求之前做些什么
    
    // 添加认证token
    const token = localStorage.getItem('auth_token');
    if (token) {
      config.headers.Authorization = `Bearer ${token}`;
    }
    
    // 添加时间戳防止缓存
    if (config.method === 'get') {
      config.params = {
        ...config.params,
        _t: Date.now()
      };
    }
    
    console.log('Request sent:', config);
    return config;
  },
  function (error) {
    // 对请求错误做些什么
    return Promise.reject(error);
  }
);

// 添加响应拦截器
axios.interceptors.response.use(
  function (response) {
    // 2xx 范围内的状态码都会触发该函数
    
    // 对响应数据做点什么
    console.log('Response received:', response);
    
    // 如果API返回统一的数据格式，可以在这里处理
    if (response.data && response.data.code === 200) {
      return response.data.data;
    }
    
    return response;
  },
  function (error) {
    // 超出 2xx 范围的状态码都会触发该函数
    
    // 统一错误处理
    if (error.response) {
      // 服务器返回了错误状态码
      const status = error.response.status;
      
      switch (status) {
        case 401:
          // 未授权，跳转到登录页
          window.location.href = '/login';
          break;
        case 403:
          // 禁止访问
          console.error('没有权限访问该资源');
          break;
        case 404:
          // 资源不存在
          console.error('请求的资源不存在');
          break;
        case 500:
          // 服务器错误
          console.error('服务器内部错误');
          break;
        default:
          console.error('请求失败:', error.message);
      }
    } else if (error.request) {
      // 请求发送了但没有收到响应
      console.error('网络错误，请检查网络连接');
    } else {
      // 其他错误
      console.error('请求配置错误:', error.message);
    }
    
    return Promise.reject(error);
  }
);

// 添加拦截器并保存引用
const requestInterceptor = axios.interceptors.request.use(
  config => config,
  error => Promise.reject(error)
);

const responseInterceptor = axios.interceptors.response.use(
  response => response,
  error => Promise.reject(error)
);

// 移除拦截器
axios.interceptors.request.eject(requestInterceptor);
axios.interceptors.response.eject(responseInterceptor);
```

### 并发请求
```js
// 使用 Promise.all
async function fetchAllData() {
  try {
    const [usersResponse, postsResponse, commentsResponse] = await Promise.all([
      axios.get('/api/users'),
      axios.get('/api/posts'),
      axios.get('/api/comments')
    ]);
    
    return {
      users: usersResponse.data,
      posts: postsResponse.data,
      comments: commentsResponse.data
    };
  } catch (error) {
    console.error('One of the requests failed:', error);
    throw error;
  }
}
```

### 文件上传
```js
// 单文件上传
const formData = new FormData();
formData.append('file', fileInput.files[0]);
formData.append('description', 'File description');

axios.post('/api/upload', formData, {
  headers: {
    'Content-Type': 'multipart/form-data'
  },
  onUploadProgress: progressEvent => {
    const percentCompleted = Math.round(
      (progressEvent.loaded * 100) / progressEvent.total
    );
    console.log(`Upload Progress: ${percentCompleted}%`);
  }
})
.then(response => {
  console.log('Upload successful:', response.data);
});

//多文件上传
const formData = new FormData();

// 添加多个文件
for (let i = 0; i < files.length; i++) {
  formData.append('files', files[i]);
}

axios.post('/api/upload/multiple', formData, {
  headers: {
    'Content-Type': 'multipart/form-data'
  }
});
```
